export function renderPropsTable(entries, showDefault, label) {
  if (entries.length === 0) return null;

return (

<div className="sb-table-scroll" role="region" aria-label={label} tabIndex={0}>
  <table>
    <thead>
      <tr>
        <th>Property</th>
        <th>Type</th>
        {showDefault ? <th>Default</th> : null}
        <th>Description</th>
      </tr>
    </thead>
    <tbody>
      {entries.map(([key, prop]) => (
        <tr key={key}>
          <td>{prop.twigProp ? <code>{prop.twigProp}</code> : '—'}</td>
          <td>
            <code>{prop.type || '—'}</code>
          </td>
          {showDefault ? (
            <td>
              <code>
                {prop.default !== undefined ? String(prop.default) : '—'}
              </code>
            </td>
          ) : null}
          <td>{prop.detail || prop.description || '—'}</td>
        </tr>
      ))}
    </tbody>
  </table>
</div>
); }

export function twigPropsTable(props, filter) {
  const entries = Object.entries(props).filter(([, p]) => {
    if (p.twigOnly) return false;
    if (filter === 'required') return p.required;
    if (filter === 'optional') return !p.required;
    return true;
  });
  const labels = {
    required: 'Required properties',
    optional: 'Optional properties',
  };
  return renderPropsTable(entries, true, labels[filter] || 'Properties');
}

export function twigOnlyPropsTable(props) {
  const entries = Object.entries(props).filter(([, p]) => p.twigOnly);
  return renderPropsTable(entries, false, 'Twig-only properties');
}
